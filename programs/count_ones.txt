START: IF 1 GOTO MARK           # 0
IF 0 GOTO NEXT                  # 1
HALT                            # 2
NEXT: RIGHT                     # 3
GOTO START                      # 4
MARK: WRITE X                   # 5
FIND_SEPARATOR: RIGHT
IF _ GOTO ENTER_COUNT
GOTO FIND_SEPARATOR
ENTER_COUNT: RIGHT
FIND_END: IF _ GOTO APPEND
RIGHT
GOTO FIND_END
APPEND: WRITE 1
RETURN_SEPARATOR: LEFT
IF _ GOTO RETURN_MARK
GOTO RETURN_SEPARATOR
RETURN_MARK: LEFT
IF X GOTO RESTORE
GOTO RETURN_MARK
RESTORE: WRITE 1
RIGHT
GOTO START
